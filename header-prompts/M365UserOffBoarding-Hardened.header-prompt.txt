function Invoke-SafeAction {
    param(
        [string]$Action,
        [scriptblock]$Code
    )

    if ($WhatIf) {
        Write-Host "[WHATIF] $Action" -ForegroundColor Yellow
        Write-AuditLog "[WHATIF] $Action"
        return "WhatIf"
    }

    try {
        & $Code
        Write-AuditLog "[SUCCESS] $Action"
        return "Success"
    }
    catch {
        Write-AuditLog "[FAILED] $Action | $($_.Exception.Message)"
        return "Failed"
    }
}
