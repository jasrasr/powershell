function Get-ModuleInstalledStatus {
    param (
        [string]$ModuleName,
        [string]$MinVersion
    )

    $module = Get-Module -ListAvailable -Name $ModuleName | Where-Object { $_.Version -ge (New-Object Version $MinVersion) }
    return $null -ne $module
}
