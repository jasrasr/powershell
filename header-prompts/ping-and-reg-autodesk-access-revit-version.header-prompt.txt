function Get-FirstPresentValue {
    param(
        [object]$Entry,
        [string[]]$PropertyNames
    )
    foreach ($p in $PropertyNames) {
        if ($Entry.PSObject.Properties.Name -contains $p) {
            $v = $Entry.$p
            if ($null -ne $v -and ($v -isnot [string] -or $v.Trim() -ne '')) {
                return $v
            }
        }
    }
    return $null
}
