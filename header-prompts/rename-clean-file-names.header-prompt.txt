function Get-CleanFileName {
    param ([string] $FileName)

    $name = [System.IO.Path]::GetFileNameWithoutExtension($FileName)
    $ext  = [System.IO.Path]::GetExtension($FileName)

    $clean = $name

    do {
        $previous = $clean

        $clean = $clean -replace '~', ''
        $clean = $clean -replace "'", ''
        $clean = $clean.Trim()

        while ($clean -match '  ') {
            $clean = $clean -replace '  ', ' '
        }

        $clean = $clean -replace '^_+', ''
        $clean = $clean -replace '_+$', ''

    } while ($clean -ne $previous)

    return "$clean$ext"
}
