<#
.SYNOPSIS
    Byte ↔ Bit Converter GUI with Download/Upload Simulation
.DESCRIPTION
    Converts between bytes and bits (KB/MB/GB/TB) and simulates both download
    and upload speeds using colored progress bars and directional animation.
.AUTHOR
    Jason Lamb (jasr.me)
.REVISION
    4.0.1
.CREATED
    2025-10-29
.CHANGES
    4.0.1 - Fixed Add-Type compile errors on PowerShell 7 by passing
            -ReferencedAssemblies for WinForms/Drawing. Guarded type definition
            so the script is safely re-runnable in the same session.
    4.0   - Initial release of dual-mode (download/upload) simulator with
            colored progress bar.
#>

Add-Type -AssemblyName System.Windows.Forms
Add-Type -AssemblyName System.Drawing

# --- Custom colored ProgressBar class ---
# Guard so re-running in the same PS session doesn't error on "type already exists".
# -ReferencedAssemblies is required on PowerShell 7+ — WinForms/Drawing are not auto-referenced.
if (-not ('ColorProgressBar' -as [type])) {
    Add-Type -ReferencedAssemblies System.Windows.Forms, System.Drawing -TypeDefinition @"
using System;
using System.Windows.Forms;
using System.Drawing;
public class ColorProgressBar : ProgressBar
{
    public Color BarColor = Color.LimeGreen;
    protected override void OnPaint(PaintEventArgs e)
    {
        Rectangle rect = e.ClipRectangle;
        if (ProgressBarRenderer.IsSupported)
            ProgressBarRenderer.DrawHorizontalBar(e.Graphics, rect);
        rect.Inflate(-3, -3);
        int width = (int)((float)Value / Maximum * rect.Width);
        if (width <= 0) width = 1;
        using (SolidBrush brush = new SolidBrush(BarColor))
        {
            e.Graphics.FillRectangle(brush, rect.X, rect.Y, width, rect.Height);
        }
    }
}
"@
}
